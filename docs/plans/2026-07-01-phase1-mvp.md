> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** 手机通过 Bridge 连接 OpenCode，实现聊天、会话管理、工具审批核心 MVP

**Architecture:** 三层架构：手机(React Native WS) → Bridge(Node.js + ws + JWT) → OpenCode(HTTP + SSE)。Bridge 持有唯一 `OpencodeClient` 实例，SSE 事件流实时转为 WS event 帧推送手机。

**Tech Stack:** Bridge: Node.js 18+ / TypeScript / ws / jsonwebtoken / @opencode-ai/sdk v2; Mobile: React Native / TypeScript / Zustand / WebSocket

**SDD 遵守原则:** 所有接口方法、参数、事件名、帧格式严格遵循 `docs/03-architecture-design.md`。`project.setup` 为新增 Phase 1 方法（审批时已确认）。

**Step 1: 实现 ws.ts**

```typescript
import { WebSocketServer, WebSocket } from "ws"
import { IncomingMessage } from "http"
import { verifyToken } from "./auth"
import { handleFrame } from "./router"

const connections = new Map<string, WebSocket>()

export function createWSServer(port: number): WebSocketServer {
  const wss = new WebSocketServer({ port })

  wss.on("connection", (ws: WebSocket, req: IncomingMessage) => {
    const url = new URL(req.url || "/", `http://${req.headers.host}`)
    const token = url.searchParams.get("token") || ""

    // JWT 验证
    const payload = verifyToken(token)
    if (!payload) {
      ws.send(JSON.stringify({ type: "res", id: "0", ok: false, error: "unauthorized" }))
      ws.close(4001, "unauthorized")
      return
    }

    const connID = `conn_${Date.now()}_${Math.random().toString(36).slice(2, 8)}`
    connections.set(connID, ws)

    console.log(`[WS] 新连接: ${connID}, user: ${payload.sub}`)

    ws.on("message", (data: Buffer) => {
      try {
        const frame = JSON.parse(data.toString())
        handleFrame(connID, ws, frame, payload)
      } catch (err) {
        ws.send(JSON.stringify({ type: "res", id: "0", ok: false, error: "invalid json" }))
      }
    })

    ws.on("close", () => {
      connections.delete(connID)
      console.log(`[WS] 断开: ${connID}`)
    })

    ws.on("error", (err) => {
      console.error(`[WS] 错误: ${connID}`, err.message)
      connections.delete(connID)
    })
  })

  console.log(`[WS] 服务器启动于端口 ${port}`)
  return wss
}
```

```typescript
import jwt from "jsonwebtoken"

const SECRET = process.env.JWT_SECRET || "dev-secret-change-in-production"
const EXPIRES_IN = "24h"

export interface TokenPayload {
  sub: string
  role: string
}

export function signToken(payload: { sub: string; role: string }): string {
  return jwt.sign(payload, SECRET, { expiresIn: EXPIRES_IN })
}

export function verifyToken(token: string): TokenPayload | null {
  try {
    const decoded = jwt.verify(token, SECRET) as TokenPayload
    return decoded
  } catch {
    return null
  }
}

export function generateToken(): { token: string; expiresIn: number } {
  const token = signToken({ sub: "mobile-user", role: "user" })
  return { token, expiresIn: 24 * 60 * 60 }
}

// auth.login handler（直接实现）
export function handleLogin(params: { password?: string }): { token: string; expiresIn: number } | { error: string } {
  const envPassword = process.env.BRIDGE_PASSWORD
  if (envPassword && params.password !== envPassword) {
    return { error: "invalid password" }
  }
  return generateToken()
}

// auth.refresh handler
export function handleRefresh(): { token: string; expiresIn: number } {
  return generateToken()
}

// auth.logout handler
export function handleLogout(): { ok: true } {
  return { ok: true }
}
```

```typescript
import { WebSocket } from "ws"
import { TokenPayload } from "./auth"
import { handleLogin, handleRefresh, handleLogout } from "./auth"
import { setupProject, getCurrentProject } from "../state/project"
import { getOpenCodeBackend } from "../adapters/OpenCodeAdapter"

interface Frame {
  type: string
  id: string
  method?: string
  params?: Record<string, unknown>
}

type Handler = (params: any, payload: TokenPayload) => Promise<any> | any

const handlers = new Map<string, Handler>()

export function registerHandler(method: string, handler: Handler): void {
  handlers.set(method, handler)
}

export async function handleFrame(connID: string, ws: WebSocket, frame: Frame, payload: TokenPayload): Promise<void> {
  if (frame.type !== "req") {
    ws.send(JSON.stringify({ type: "res", id: frame.id || "0", ok: false, error: "invalid frame type" }))
    return
  }

  const method = frame.method || ""
  const handler = handlers.get(method)

  if (!handler) {
    ws.send(JSON.stringify({ type: "res", id: frame.id, ok: false, error: `unknown method: ${method}` }))
    return
  }

  try {
    const result = await handler(frame.params || {}, payload)
    ws.send(JSON.stringify({ type: "res", id: frame.id, ok: true, payload: result }))
  } catch (err: any) {
    ws.send(JSON.stringify({ type: "res", id: frame.id, ok: false, error: err.message || "internal error" }))
  }
}

// ===== 注册标准 handler（根据 method 前缀路由）=====

// 认证（直接实现）
registerHandler("auth.login", (params) => handleLogin(params))
registerHandler("auth.refresh", () => handleRefresh())

// 健康检查（直接实现）
registerHandler("health.ping", () => ({ ok: true }))

// 项目设置（直接实现）
registerHandler("project.setup", async (params) => setupProject(params.directory))
registerHandler("project.current", async () => getCurrentProject())

// ===== 代理方法注册（通过 OpenCodeAdapter）=====

registerHandler("session.list", async (params) => getOpenCodeBackend().sdk?.session.list(params))
registerHandler("session.get", async (params) => getOpenCodeBackend().sdk?.session.get(params))
registerHandler("session.delete", async (params) => getOpenCodeBackend().sdk?.session.delete(params))
registerHandler("session.rename", async (params) => getOpenCodeBackend().sdk?.session.update(params))
registerHandler("session.messages", async (params) => getOpenCodeBackend().sdk?.session.messages(params))
registerHandler("session.status", async () => getOpenCodeBackend().sdk?.session.status({}))

// message.*
```

**Step 1: project.ts**（管理活跃目录、SDK client、SSE 生命周期）

```typescript
import { createOpencodeClient, OpencodeClient } from "@opencode-ai/sdk/v2"
import { broadcastToAll } from "../server/ws"

interface ProjectState {
  activeDirectory: string | null
  currentProject: { name?: string } | null
  sdk: OpencodeClient | null
  sseAbort: AbortController | null
  sseLoop: Promise<void> | null
  isSwitching: boolean
}

const state: ProjectState = {
  activeDirectory: null,
  currentProject: null,
  sdk: null,
  sseAbort: null,
  sseLoop: null,
  isSwitching: false,
}

const BASE_URL = process.env.OPENCODE_URL || "http://localhost:4096"

export function getOpenCodeBackend() {
  return { sdk: state.sdk }
}

async function startSSE(sdk: OpencodeClient, signal: AbortSignal): Promise<void> {
  while (true) {
    if (signal.aborted) break
    try {
      const events = await sdk.global.event({ signal, sseMaxRetryAttempts: 0 })
      for await (const event of events.stream) {
        if (signal.aborted) break
        broadcastToAll({ type: "event", event: event.type, data: event.data })
      }
    } catch (err: any) {
      if (signal.aborted) break
      console.error("[SSE] 错误:", err.message)
    }
    await new Promise(r => setTimeout(r, 3000))
  }
}

export async function setupProject(directory: string): Promise<{ directory: string; project?: { name?: string } } | { error: string }> {
  if (state.isSwitching) return { error: "already switching" }
  state.isSwitching = true

  try {
    // 清理旧状态
    state.sseAbort?.abort()
    state.sdk = null
    state.sseAbort = null
    state.sseLoop = null

    // 创建新 client
    const sdk = createOpencodeClient({ baseUrl: BASE_URL, directory })
    state.sdk = sdk
    state.activeDirectory = directory

    // 启动 SSE
    const abort = new AbortController()
    state.sseAbort = abort
    state.sseLoop = startSSE(sdk, abort.signal)

    // 读取项目元信息
    state.currentProject = { name: directory.split("/").pop() || "unknown" }

    state.isSwitching = false

    // 广播 project.changed
    setTimeout(() => {
      broadcastToAll({
        type: "event",
        event: "project.changed",
        data: { directory: state.activeDirectory, project: state.currentProject },
      })
    }, 0)

    return { directory: state.activeDirectory, project: state.currentProject }
  } catch (err: any) {
    state.isSwitching = false
    return { error: err.message || "setup failed" }
  }
}

export function getCurrentProject(): { directory: string | null; project: { name?: string } | null } {
  return { directory: state.activeDirectory, project: state.currentProject }
}
```

```typescript
import { WebSocketServer, WebSocket } from "ws"

const connections = new Map<string, WebSocket>()

export function broadcastToAll(frame: any): void {
  const msg = JSON.stringify(frame)
  for (const [id, ws] of connections) {
    if (ws.readyState === WebSocket.OPEN) {
      ws.send(msg)
    }
  }
}
```

```typescript
// BridgeClient.ts — 完整实现
let uuidCounter = 0
function genId(): string {
  return `req_${++uuidCounter}_${Date.now()}`
}

export class BridgeClient {
  private ws: WebSocket | null = null
  private pending = new Map<string, {
    resolve: (v: any) => void
    reject: (e: Error) => void
    timeout: ReturnType<typeof setTimeout>
  }>()
  private handlers = new Map<string, (data: any) => void>()
  private reconnectTimer: ReturnType<typeof setTimeout> | null = null
  private url = ""
  private token = ""
  private _onStatusChange?: (status: string) => void

  set onStatusChange(cb: (status: string) => void) { this._onStatusChange = cb }

  connect(url: string, token: string): void {
    this.url = url
    this.token = token
    this._onStatusChange?.("connecting")

    this.ws = new WebSocket(`${url}/ws?token=${token}`)

    this.ws.onopen = () => {
      console.log("[BridgeClient] 连接成功")
      this._onStatusChange?.("connected")
    }

    this.ws.onmessage = (event: MessageEvent) => {
      try {
        const msg = JSON.parse(event.data as string)
        if (msg.type === "res") {
          const pending = this.pending.get(msg.id)
          if (pending) {
            if (msg.ok) pending.resolve(msg.payload)
            else pending.reject(new Error(msg.error || "request failed"))
            clearTimeout(pending.timeout)
            this.pending.delete(msg.id)
          }
        } else if (msg.type === "event") {
          const handler = this.handlers.get(msg.event)
          if (handler) handler(msg.data)
        }
      } catch (e) {
        console.error("[BridgeClient] parse error:", e)
      }
    }

    this.ws.onclose = () => {
      console.warn("[BridgeClient] 断开，5s 后重连")
      this._onStatusChange?.("disconnected")
      this.reconnectTimer = setTimeout(() => this.connect(this.url, this.token), 5000)
    }

    this.ws.onerror = (e) => {
      console.error("[BridgeClient] 错误:", e)
      this._onStatusChange?.("error")
    }
  }

  disconnect(): void {
    if (this.reconnectTimer) clearTimeout(this.reconnectTimer)
    this.ws?.close()
    this.ws = null
    this._onStatusChange?.("disconnected")
  }

  async call(method: string, params: any = {}): Promise<any> {
    const id = genId()
    return new Promise((resolve, reject) => {
      const timeout = setTimeout(() => {
        this.pending.delete(id)
        reject(new Error("请求超时"))
      }, 30000)
      this.pending.set(id, { resolve, reject, timeout })
      this.ws?.send(JSON.stringify({ type: "req", id, method, params }))
    })
  }

  on(event: string, handler: (data: any) => void): void {
    this.handlers.set(event, handler)
  }

  off(event: string): void {
    this.handlers.delete(event)
  }

  // === 认证 ===
  login(password?: string) { return this.call("auth.login", { password }) }
  refreshToken() { return this.call("auth.refresh") }
  logout() { return this.call("auth.logout") }

  // === 项目 ===
  setupProject(directory: string) { return this.call("project.setup", { directory }) }
  getCurrentProject() { return this.call("project.current") }

  // === 会话 ===
  createSession(params?: { title?: string; agent?: string }) { return this.call("session.create", params) }
  listSessions(params?: { scope?: string; search?: string; limit?: number }) { return this.call("session.list", params) }
  getSession(sessionID: string) { return this.call("session.get", { sessionID }) }
  deleteSession(sessionID: string) { return this.call("session.delete", { sessionID }) }
  renameSession(sessionID: string, title: string) { return this.call("session.rename", { sessionID, title }) }
  getSessionMessages(sessionID: string, params?: { limit?: number; before?: string }) {
    return this.call("session.messages", { sessionID, ...params })
  }

  // === 消息 ===
  sendMessage(sessionID: string, parts: any[]) { return this.call("message.send", { sessionID, parts }) }
  abortSession(sessionID: string) { return this.call("message.abort", { sessionID }) }

  // === 权限 ===
  replyPermission(requestID: string, reply?: "once" | "always" | "reject", message?: string) {
    return this.call("permission.reply", { requestID, reply, message })
  }

  // === 配置 ===
  getConfig() { return this.call("config.get") }
  getProviders() { return this.call("config.providers") }
  getAgents() { return this.call("config.agents") }

  // === 健康 ===
  ping() { return this.call("health.ping") }
}
```

```typescript
import { create } from "zustand"

interface Message {
  id: string
  role: "user" | "agent" | "tool"
  content: string
  timestamp: number
}

interface MessagesState {
  bySession: Record<string, Message[]>
  setMessages: (sessionID: string, msgs: Message[]) => void
  appendMessage: (sessionID: string, msg: Message) => void
  appendDelta: (sessionID: string, messageID: string, delta: string) => void
}

export const useMessagesStore = create<MessagesState>((set) => ({
  bySession: {},
  setMessages: (sessionID, msgs) => set((state) => ({
    bySession: { ...state.bySession, [sessionID]: msgs },
  })),
  appendMessage: (sessionID, msg) => set((state) => ({
    bySession: {
      ...state.bySession,
      [sessionID]: [...(state.bySession[sessionID] || []), msg],
    },
  })),
  appendDelta: (sessionID, messageID, delta) => set((state) => {
    const msgs = state.bySession[sessionID] || []
    const updated = msgs.map(m =>
      m.id === messageID ? { ...m, content: m.content + delta } : m
    )
    return { bySession: { ...state.bySession, [sessionID]: updated } }
  }),
}))
```
